using UnityEngine; [RequireComponent(typeof(Rigidbody))] public class CarController : MonoBehaviour { [Header("References")] public Rigidbody rb; public Transform centerOfMass; public WheelCollider frontLeftWheel; public WheelCollider frontRightWheel; public WheelCollider rearLeftWheel; public WheelCollider rearRightWheel; [Header("Driving")] public float maxMotorTorque = 1500f; public float maxBrakeTorque = 3000f; public float maxSteerAngle = 28f; public float maxSpeed = 45f; [Header("Drift")] public float grip = 0.9f; public float driftGrip = 0.55f; public float steerAssist = 0.7f; private float throttleInput; private float steeringInput; private float brakeInput; private float currentSpeed; private void Start() { if (rb == null) rb = GetComponent(); if (centerOfMass != null) rb.centerOfMass = centerOfMass.localPosition; rb.drag = 0.05f; rb.angularDrag = 0.4f; } private void Update() { throttleInput = Input.GetAxis("Vertical"); steeringInput = Input.GetAxis("Horizontal"); brakeInput = Input.GetKey(KeyCode.Space) ? 1f : 0f; float steerTarget = steeringInput * maxSteerAngle; frontLeftWheel.steerAngle = steerTarget; frontRightWheel.steerAngle = steerTarget; float motorTorque = throttleInput * maxMotorTorque; frontLeftWheel.motorTorque = motorTorque; frontRightWheel.motorTorque = motorTorque; float brakeTorque = brakeInput * maxBrakeTorque; frontLeftWheel.brakeTorque = brakeTorque; frontRightWheel.brakeTorque = brakeTorque; rearLeftWheel.brakeTorque = brakeTorque; rearRightWheel.brakeTorque = brakeTorque; } private void FixedUpdate() { currentSpeed = rb.velocity.magnitude; if (currentSpeed > maxSpeed) { rb.velocity = rb.velocity.normalized * maxSpeed; } // 使用车头方向和侧向修正 Vector3 forward = transform.forward; Vector3 right = transform.right; Vector3 localVelocity = transform.InverseTransformDirection(rb.velocity); float lateralSpeed = localVelocity.x; float driftFactor = Input.GetKey(KeyCode.LeftShift) ? driftGrip : grip; Vector3 adjustedVelocity = rb.velocity; adjustedVelocity -= right * lateralSpeed * (1f - driftFactor) * 0.4f; rb.velocity = Vector3.Lerp(rb.velocity, adjustedVelocity, 0.1f); } }
